find_package(Threads REQUIRED)
add_library(art_cuda STATIC deconvolution.cu resize.cu)
set_target_properties(art_cuda PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED ON
    POSITION_INDEPENDENT_CODE ON)
target_compile_options(art_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(art_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-ffp-contract=off>)
endif()
option(ART_CUDA_STATIC_RUNTIME "Link CUDA runtime statically for standalone packages" OFF)
if(ART_CUDA_STATIC_RUNTIME)
    target_link_libraries(art_cuda PUBLIC CUDA::cudart_static Threads::Threads)
else()
    target_link_libraries(art_cuda PUBLIC CUDA::cudart Threads::Threads)
endif()
if(OpenMP_CXX_FOUND AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_definitions(art_cuda PRIVATE ART_CUDA_OPENMP)
    target_compile_options(art_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=${OpenMP_CXX_FLAGS}>)
    target_link_libraries(art_cuda PUBLIC OpenMP::OpenMP_CXX)
endif()
