cmake_minimum_required(VERSION 3.18)
project(art_gpu_pipeline_tests LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CUDA_STANDARD 14)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIBMM REQUIRED IMPORTED_TARGET glibmm-2.4)
pkg_check_modules(LCMS REQUIRED IMPORTED_TARGET lcms2)
add_library(haar_cuda STATIC haar.cu)
target_compile_options(haar_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
target_link_libraries(haar_cuda PUBLIC CUDA::cudart)
add_executable(haar_test haar_test.cc)
target_compile_options(haar_test PRIVATE -march=native -ffp-contract=off)
target_link_libraries(haar_test PRIVATE haar_cuda OpenMP::OpenMP_CXX PkgConfig::GLIBMM PkgConfig::LCMS)
enable_testing()
add_test(NAME haar_equivalence COMMAND haar_test)
set_tests_properties(haar_equivalence PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=28")
