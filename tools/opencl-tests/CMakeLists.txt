cmake_minimum_required(VERSION 3.18)
project(art_opencl_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(OpenCL REQUIRED)
find_package(OpenMP REQUIRED)
find_package(TIFF REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIBMM REQUIRED IMPORTED_TARGET glibmm-2.4)
pkg_check_modules(LCMS REQUIRED IMPORTED_TARGET lcms2)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/kernels.cl" KERNELS)
configure_file(kernels_source.h.in kernels_source.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS kernels.cl)
add_library(opencl_experiment STATIC backend.cc)
target_include_directories(opencl_experiment PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(opencl_experiment PUBLIC OpenCL::OpenCL OpenMP::OpenMP_CXX)
add_library(cpu_reference STATIC ../../rtengine/lanczos.cc ../../rtengine/deconvolution.cc ../../rtengine/gauss.cc)
target_link_libraries(cpu_reference PUBLIC OpenMP::OpenMP_CXX PkgConfig::GLIBMM PkgConfig::LCMS)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cpu_reference PRIVATE -ffp-contract=off -march=native)
    target_compile_options(opencl_experiment PRIVATE -ffp-contract=off -march=native)
endif()
foreach(test deconvolution resize)
    add_executable(${test}_test ../cuda-tests/${test}_test.cc)
    target_compile_definitions(${test}_test PRIVATE ART_TEST_OPENCL)
    target_link_libraries(${test}_test PRIVATE opencl_experiment cpu_reference TIFF::TIFF)
endforeach()
option(ART_COMPARE_CUDA "Also measure the existing CUDA backend in the same process" OFF)
add_executable(gpu_benchmark benchmark.cc)
target_link_libraries(gpu_benchmark PRIVATE opencl_experiment cpu_reference TIFF::TIFF)
if(ART_COMPARE_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_subdirectory(../../rtengine/cuda cuda)
    target_compile_definitions(gpu_benchmark PRIVATE ART_COMPARE_CUDA)
    target_link_libraries(gpu_benchmark PRIVATE art_cuda)
endif()
enable_testing()
# Use a genuinely empty ICD directory, not a mocked successful device selection.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/empty-icd")
foreach(test deconvolution resize)
    add_test(NAME opencl_${test}_equivalence COMMAND ${test}_test)
    set_tests_properties(opencl_${test}_equivalence PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=28")
    add_test(NAME opencl_${test}_no_device COMMAND ${test}_test --no-device)
    set_tests_properties(opencl_${test}_no_device PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=28;OCL_ICD_VENDORS=${CMAKE_CURRENT_BINARY_DIR}/empty-icd")
endforeach()
