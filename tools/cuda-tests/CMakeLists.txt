cmake_minimum_required(VERSION 3.18)
project(art_cuda_tests LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 14)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED)
find_package(TIFF REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIBMM REQUIRED IMPORTED_TARGET glibmm-2.4)
pkg_check_modules(LCMS REQUIRED IMPORTED_TARGET lcms2)
add_subdirectory(../../rtengine/cuda cuda)
add_executable(deconvolution_test deconvolution_test.cc
    ../../rtengine/deconvolution.cc ../../rtengine/gauss.cc)
target_link_libraries(deconvolution_test PRIVATE art_cuda OpenMP::OpenMP_CXX PkgConfig::GLIBMM PkgConfig::LCMS)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(deconvolution_test PRIVATE -ffp-contract=off -march=native)
endif()
enable_testing()
add_executable(compare_tiff compare_tiff.cc)
target_link_libraries(compare_tiff PRIVATE TIFF::TIFF)
add_test(NAME deconvolution_equivalence COMMAND deconvolution_test)
set_tests_properties(deconvolution_equivalence PROPERTIES ENVIRONMENT "ART_CUDA=1;OMP_NUM_THREADS=28")
add_test(NAME deconvolution_no_device COMMAND deconvolution_test --no-device)
set_tests_properties(deconvolution_no_device PROPERTIES ENVIRONMENT "ART_CUDA=1;CUDA_VISIBLE_DEVICES=;OMP_NUM_THREADS=28")
add_executable(resize_test resize_test.cc ../../rtengine/lanczos.cc)
target_link_libraries(resize_test PRIVATE art_cuda OpenMP::OpenMP_CXX PkgConfig::GLIBMM PkgConfig::LCMS TIFF::TIFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(resize_test PRIVATE -ffp-contract=off -march=native)
endif()
add_test(NAME resize_equivalence COMMAND resize_test)
set_tests_properties(resize_equivalence PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=28")
add_test(NAME resize_no_device COMMAND resize_test --no-device)
set_tests_properties(resize_no_device PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=;OMP_NUM_THREADS=28")

add_test(NAME deconvolution_recursive COMMAND deconvolution_test --recursive)
set_tests_properties(deconvolution_recursive PROPERTIES ENVIRONMENT "ART_CUDA=1;ART_CUDA_RECURSIVE=1;OMP_NUM_THREADS=28")

add_executable(hybrid_test hybrid_test.cc ../../rtengine/deconvolution.cc ../../rtengine/gauss.cc)
target_compile_definitions(hybrid_test PRIVATE ART_USE_CUDA)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(hybrid_test PRIVATE -ffp-contract=off -march=native)
endif()
target_link_libraries(hybrid_test PRIVATE art_cuda OpenMP::OpenMP_CXX PkgConfig::GLIBMM PkgConfig::LCMS)
add_test(NAME hybrid_equivalence COMMAND hybrid_test)
set_tests_properties(hybrid_equivalence PROPERTIES ENVIRONMENT "ART_HYBRID=1;ART_CUDA=1;ART_CUDA_RECURSIVE=1;OMP_NUM_THREADS=28")
add_test(NAME hybrid_fallback COMMAND hybrid_test)
set_tests_properties(hybrid_fallback PROPERTIES ENVIRONMENT "ART_HYBRID=1;ART_CUDA=1;ART_CUDA_RECURSIVE=1;CUDA_VISIBLE_DEVICES=;OMP_NUM_THREADS=28")
add_test(NAME cuda_pair_equivalence COMMAND hybrid_test)
set_tests_properties(cuda_pair_equivalence PROPERTIES ENVIRONMENT "ART_GPU_PAIR=1;ART_CUDA=1;ART_CUDA_RECURSIVE=1;OMP_NUM_THREADS=28")
