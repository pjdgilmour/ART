cmake_minimum_required(VERSION 3.12)
project(art_performance_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
add_library(policy ../../rtengine/cpupolicy.cc ../../rtengine/rawconvert.cc)
target_compile_options(policy PRIVATE -O3 -ffp-contract=off)
target_link_libraries(policy PUBLIC Threads::Threads OpenMP::OpenMP_CXX)
add_executable(policy_test policy_test.cc)
target_link_libraries(policy_test PRIVATE policy)
add_executable(rawconvert_bench rawconvert_bench.cc)
target_link_libraries(rawconvert_bench PRIVATE policy)
enable_testing()
add_test(NAME policy_dynamic COMMAND policy_test)
set_tests_properties(policy_dynamic PROPERTIES ENVIRONMENT "ART_SCHEDULER=dynamic")
add_test(NAME policy_groups COMMAND policy_test)
set_tests_properties(policy_groups PROPERTIES ENVIRONMENT "ART_SCHEDULER=groups")
add_executable(review_test review_test.cc)
target_link_libraries(review_test PRIVATE Threads::Threads OpenMP::OpenMP_CXX)
add_test(NAME review_concurrency COMMAND review_test)
set_tests_properties(review_concurrency PROPERTIES TIMEOUT 10)

add_executable(review_serial_test review_test.cc)
target_link_libraries(review_serial_test PRIVATE Threads::Threads)
add_test(NAME review_without_openmp COMMAND review_serial_test)
set_tests_properties(review_without_openmp PROPERTIES TIMEOUT 10)
